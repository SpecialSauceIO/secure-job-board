      const cost_job_ai = parseFloat(document.getElementById('cost-job-ai').value) || 0.001;

      const cost_fixed = parseFloat(document.getElementById('cost-fixed').value) || 46;

      const stripe_pct = (parseFloat(document.getElementById('stripe-pct').value) || 2.9) / 100;


      // --- Calculations ---

      // Inbound Revenues

      const c2_rev = c2_price * c2_members;

      const c3_rev = c3_price * c3_members;

      const cand_rev = c2_rev + c3_rev;


      const e1_rev = e1_price * e1_members;

      const e2_rev = e2_price * e2_members;

      const e3_rev = e3_price * e3_members;

      const emp_rev = e1_rev + e2_rev + e3_rev;


      const gross_mrr = cand_rev + emp_rev;

      const arr = gross_mrr * 12;

      const total_paying_subs = c2_members + c3_members + e1_members + e2_members + e3_members;


      // Payment Gateway Processing Fees (2.9% + $0.30 per successful paid transaction)

      const stripe_variable_fees = (gross_mrr * stripe_pct) + (total_paying_subs * 0.30);


      // Usage Outbound Costs

      const total_applications = (c1_members * c1_apps) + (c2_members * c2_apps) + (c3_members * c3_apps);

      const total_ai_match_cost = total_applications * cost_match_ai;


      const total_jobs_posted = (e1_members * e1_jobs) + (e2_members * e2_jobs) + (e3_members * e3_jobs);

      const total_ai_job_cost = total_jobs_posted * cost_job_ai;


      const total_id_verification_cost = new_verifications * cost_id_verify;


      const total_variable_costs = stripe_variable_fees + total_ai_match_cost + total_ai_job_cost + total_id_verification_cost;

      const total_all_costs = total_variable_costs + cost_fixed;


      const net_profit = gross_mrr - total_all_costs;

      const annual_profit = net_profit * 12;

      const gross_margin = gross_mrr > 0 ? ((net_profit / gross_mrr) * 100).toFixed(1) : 0;


      // --- UI Updates ---

      document.getElementById('kpi-mrr').innerText = fmtUSD(gross_mrr);

      document.getElementById('kpi-arr').innerText = `ARR: ${fmtUSD(arr)}`;

      document.getElementById('kpi-var-cost').innerText = fmtUSDFloat(total_variable_costs);

      document.getElementById('kpi-fixed-cost').innerText = fmtUSD(cost_fixed);

      document.getElementById('kpi-profit').innerText = fmtUSD(net_profit);

      document.getElementById('kpi-annual-profit').innerText = `Annual: ${fmtUSD(annual_profit)}`;

      document.getElementById('kpi-margin').innerText = `${gross_margin}%`;

      document.getElementById('kpi-active-subs').innerText = `${total_paying_subs} Paying Subs`;


      // Tier subtotals

      document.getElementById('candidate-subtotal').innerText = fmtUSD(cand_rev) + '/mo';

      document.getElementById('employer-subtotal').innerText = fmtUSD(emp_rev) + '/mo';

      document.getElementById('c2-rev').innerText = fmtUSD(c2_rev) + '/mo';

      document.getElementById('c3-rev').innerText = fmtUSD(c3_rev) + '/mo';

      document.getElementById('e1-rev').innerText = fmtUSD(e1_rev) + '/mo';

      document.getElementById('e2-rev').innerText = fmtUSD(e2_rev) + '/mo';

      document.getElementById('e3-rev').innerText = fmtUSD(e3_rev) + '/mo';


      // Ledger breakdown

      document.getElementById('breakdown-cand-rev').innerText = fmtUSD(cand_rev);

      document.getElementById('breakdown-emp-rev').innerText = fmtUSD(emp_rev);

      document.getElementById('breakdown-total-rev').innerText = fmtUSD(gross_mrr);


      document.getElementById('breakdown-id-cost').innerText = fmtUSDFloat(total_id_verification_cost);

      document.getElementById('breakdown-match-cost').innerText = fmtUSDFloat(total_ai_match_cost + total_ai_job_cost);

      document.getElementById('breakdown-stripe-fees').innerText = fmtUSDFloat(stripe_variable_fees);

      document.getElementById('breakdown-fixed-cost').innerText = fmtUSD(cost_fixed);

      document.getElementById('breakdown-total-cost').innerText = fmtUSDFloat(total_all_costs);

    }
